โค้ด Google Apps Script (Code.gs)

คัดลอกโค้ดด้านล่างนี้ไปวางในเมนู ส่วนขยาย (Extensions) > Apps Script ใน Google Sheet ของคุณ แล้วทำการ Deploy เป็น Web App โดยโค้ดนี้จะสร้างชีต Categories สำหรับเก็บหมวดหมู่ และใช้ร่วมกันทุกเครื่อง:

// ==========================================
// Google Apps Script (Code.gs)
// ระบบบริหารจัดการครุภัณฑ์ - เทศบาลนครภูเก็ต
// ==========================================

const SPREADSHEET_ID = "1mm7Jrc5DICxn_928BpImHDR7ZIg5Lc4grpvltWE1Rtc";
const SHEET_NAME = "Data";
const CATEGORY_SHEET_NAME = "Categories";
const DEFAULT_CATEGORIES = ["โต๊ะ", "เก้าอี้", "ตู้", "คอมพิวเตอร์", "พัดลม", "เครื่องปริ้น"];

function doGet(e) {
  try {
    const action = e && e.parameter ? (e.parameter.action || 'status') : 'status';
    if (action === 'read') return readAssets_();
    if (action === 'readCategories') return readCategories_();
    return responseJSON({ status: "success", message: "Phuket Municipality Asset API Ready" });
  } catch (error) {
    return responseJSON({ status: "error", message: error.toString() });
  }
}

function doPost(e) {
  try {
    let contents = {};
    if (e && e.postData && e.postData.contents) {
      try {
        contents = JSON.parse(e.postData.contents);
      } catch (jsonError) {
        contents = e.parameter || {};
        if (typeof contents.data === 'string') contents.data = JSON.parse(contents.data);
      }
    } else {
      contents = e.parameter || {};
    }

    const action = contents.action;
    if (action === 'read') return readAssets_();
    if (action === 'readCategories') return readCategories_();
    if (action === 'addCategory') return addCategory_(contents.name);
    if (action === 'deleteCategory') return deleteCategory_(contents.name);

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow(["ID", "Category", "Name", "Code", "Detail", "Budget", "Amount", "ReceivedDate", "DikaNo", "Location", "ImageUrl"]);
    }

    if (action === "save") {
      const item = contents.data || {};
      const data = sheet.getDataRange().getValues();
      let rowIndex = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(item.id)) { rowIndex = i + 1; break; }
      }
      const rowValues = [
        item.id || Utilities.getUuid(), item.category || '', item.name || '', item.code || '',
        item.detail || '', item.budget || '', Number(String(item.amount == null ? 0 : item.amount).replace(/,/g, '')) || 0, item.receivedDate || '',
        item.dikaNo || '', item.location || '', item.imageUrl || ''
      ];
      if (rowIndex > 0) sheet.getRange(rowIndex, 1, 1, rowValues.length).setValues([rowValues]);
      else sheet.appendRow(rowValues);
      return responseJSON({ status: "success", message: "Data saved successfully" });
    }

    if (action === "delete") {
      const id = contents.id;
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(id)) { sheet.deleteRow(i + 1); break; }
      }
      return responseJSON({ status: "success", message: "Data deleted" });
    }

    return responseJSON({ status: "error", message: "Unknown action" });
  } catch (error) {
    return responseJSON({ status: "error", message: error.toString() });
  }
}

function ensureCategorySheet_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(CATEGORY_SHEET_NAME);

  if (!sheet) {
    sheet = ss.insertSheet(CATEGORY_SHEET_NAME);
    sheet.getRange(1, 1).setValue('Category');

    const existing = [];
    const dataSheet = ss.getSheetByName(SHEET_NAME);
    if (dataSheet && dataSheet.getLastRow() > 1) {
      const values = dataSheet.getRange(2, 2, dataSheet.getLastRow() - 1, 1).getValues();
      values.forEach(r => {
        const name = String(r[0] || '').trim();
        if (name && !existing.includes(name)) existing.push(name);
      });
    }

    const initial = existing.length ? existing : DEFAULT_CATEGORIES.slice();
    if (initial.length) sheet.getRange(2, 1, initial.length, 1).setValues(initial.map(x => [x]));
  }

  return sheet;
}

function readCategories_() {
  const sheet = ensureCategorySheet_();
  const lastRow = sheet.getLastRow();
  const categories = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0] || '').trim()).filter(Boolean)
    : [];

  return responseJSON({ status: "success", data: categories });
}

function addCategory_(name) {
  const category = String(name || '').trim();
  if (!category) return responseJSON({ status: "error", message: "กรุณาระบุชื่อหมวดหมู่" });

  const sheet = ensureCategorySheet_();
  const lastRow = sheet.getLastRow();
  const existing = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0] || '').trim()).filter(Boolean)
    : [];

  if (existing.some(x => x.toLowerCase() === category.toLowerCase())) {
    return responseJSON({ status: "error", message: "หมวดหมู่นี้มีอยู่ในระบบแล้ว" });
  }

  sheet.getRange(sheet.getLastRow() + 1, 1).setValue(category);
  return responseJSON({ status: "success", message: "เพิ่มหมวดหมู่เรียบร้อยแล้ว", data: existing.concat([category]) });
}

function deleteCategory_(name) {
  const category = String(name || '').trim();
  if (!category) return responseJSON({ status: "error", message: "ไม่พบชื่อหมวดหมู่" });

  const sheet = ensureCategorySheet_();
  const lastRow = sheet.getLastRow();
  const values = lastRow > 1 ? sheet.getRange(2, 1, lastRow - 1, 1).getValues() : [];
  const categories = values.map(r => String(r[0] || '').trim()).filter(Boolean);

  if (categories.length <= 1) {
    return responseJSON({ status: "error", message: "ต้องมีหมวดหมู่เหลืออย่างน้อย 1 รายการ" });
  }

  // ป้องกันการลบหมวดหมู่ที่ยังมีครุภัณฑ์ใช้งานอยู่
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const dataSheet = ss.getSheetByName(SHEET_NAME);
  if (dataSheet && dataSheet.getLastRow() > 1) {
    const assetCategories = dataSheet.getRange(2, 2, dataSheet.getLastRow() - 1, 1).getValues();
    const usedCount = assetCategories.filter(r => String(r[0] || '').trim() === category).length;
    if (usedCount > 0) {
      return responseJSON({ status: "error", message: `ไม่สามารถลบหมวดหมู่ "${category}" ได้ เนื่องจากมีครุภัณฑ์ ${usedCount} รายการใช้งานหมวดหมู่นี้อยู่` });
    }
  }

  for (let i = 0; i < categories.length; i++) {
    if (categories[i].toLowerCase() === category.toLowerCase()) {
      sheet.deleteRow(i + 2);
      break;
    }
  }

  return responseJSON({ status: "success", message: "ลบหมวดหมู่เรียบร้อยแล้ว" });
}

function readAssets_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);
  const categoriesSheet = ensureCategorySheet_();

  if (!sheet) return responseJSON({ status: "success", data: [], categories: readCategoriesArray_(categoriesSheet) });

  const values = sheet.getDataRange().getValues();
  const rows = values.length <= 1 ? [] : values.slice(1).map(r => ({
    id: r[0] == null ? '' : String(r[0]),
    category: r[1] == null ? '' : String(r[1]),
    name: r[2] == null ? '' : String(r[2]),
    code: r[3] == null ? '' : String(r[3]),
    detail: r[4] == null ? '' : String(r[4]),
    budget: r[5] == null ? '' : String(r[5]),
    amount: r[6] == null ? 0 : Number(String(r[6]).replace(/,/g, '')) || 0,
    receivedDate: r[7] == null ? '' : String(r[7]),
    dikaNo: r[8] == null ? '' : String(r[8]),
    location: r[9] == null ? '' : String(r[9]),
    imageUrl: r[10] == null ? '' : String(r[10])
  })).filter(x => x.id || x.code || x.name);

  return responseJSON({ status: "success", data: rows, categories: readCategoriesArray_(categoriesSheet) });
}

function readCategoriesArray_(sheet) {
  const lastRow = sheet.getLastRow();
  return lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0] || '').trim()).filter(Boolean)
    : [];
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
